use serde::{Deserialize, Serialize};

use crate::ids::{ThreadId, TurnId};

/// What happens while an engine runs a turn. The `type` field in JSON names the event.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
#[serde(tag = "type")]
pub enum Event {
    #[serde(rename = "turn.started")]
    TurnStarted {
        thread_id: ThreadId,
        turn_id: TurnId,
    },

    /// The next piece of the reply text.
    #[serde(rename = "content.delta")]
    ContentDelta {
        thread_id: ThreadId,
        turn_id: TurnId,
        text: String,
    },

    #[serde(rename = "turn.completed")]
    TurnCompleted {
        thread_id: ThreadId,
        turn_id: TurnId,
    },

    /// The turn stopped with an error. No other event follows it for this turn.
    #[serde(rename = "runtime.error")]
    RuntimeError {
        thread_id: ThreadId,
        turn_id: TurnId,
        code: ErrorCode,
        message: String,
    },
}

/// Why a turn stopped. Programs decide on the code; people read the message.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum ErrorCode {
    DriverUnavailable,
    EngineFailed,
    Internal,
}

#[cfg(test)]
mod tests {
    use serde_json::{Value, json};

    use super::{ErrorCode, Event};
    use crate::ids::{ThreadId, TurnId};

    fn assert_json(event: Event, expected: Value) {
        assert_eq!(serde_json::to_value(&event).unwrap(), expected);
        assert_eq!(serde_json::from_value::<Event>(expected).unwrap(), event);
    }

    #[test]
    fn turn_started_json() {
        assert_json(
            Event::TurnStarted {
                thread_id: ThreadId::new("thread-1"),
                turn_id: TurnId::new("turn-1"),
            },
            json!({ "type": "turn.started", "thread_id": "thread-1", "turn_id": "turn-1" }),
        );
    }

    #[test]
    fn content_delta_json() {
        assert_json(
            Event::ContentDelta {
                thread_id: ThreadId::new("thread-1"),
                turn_id: TurnId::new("turn-1"),
                text: "Hel".to_owned(),
            },
            json!({
                "type": "content.delta",
                "thread_id": "thread-1",
                "turn_id": "turn-1",
                "text": "Hel",
            }),
        );
    }

    #[test]
    fn turn_completed_json() {
        assert_json(
            Event::TurnCompleted {
                thread_id: ThreadId::new("thread-1"),
                turn_id: TurnId::new("turn-1"),
            },
            json!({ "type": "turn.completed", "thread_id": "thread-1", "turn_id": "turn-1" }),
        );
    }

    #[test]
    fn runtime_error_json() {
        assert_json(
            Event::RuntimeError {
                thread_id: ThreadId::new("thread-1"),
                turn_id: TurnId::new("turn-1"),
                code: ErrorCode::EngineFailed,
                message: "the engine closed the connection".to_owned(),
            },
            json!({
                "type": "runtime.error",
                "thread_id": "thread-1",
                "turn_id": "turn-1",
                "code": "engine_failed",
                "message": "the engine closed the connection",
            }),
        );
    }
}
